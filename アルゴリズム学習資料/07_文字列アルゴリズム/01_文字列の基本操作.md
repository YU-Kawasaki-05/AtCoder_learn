# 文字列の基本操作

文字列操作はプログラミングコンテストでよく出題されるテーマの一つです。この資料では、文字列の基本的な操作とそのPythonでの実装方法を説明します。

## 1. 文字列の基本

### 1.1 文字列の作成と参照

```python
# 文字列の作成
s = "Hello, World!"
t = 'Hello, Python!'

# 文字列の長さ
len(s)  # 13

# インデックスによる参照
s[0]   # 'H'
s[7]   # 'W'
s[-1]  # '!' （負のインデックスは末尾からの位置）

# スライシング
s[0:5]    # "Hello"
s[7:]     # "World!"
s[:5]     # "Hello"
s[::2]    # "Hlo ol!" （2つ飛ばし）
s[::-1]   # "!dlroW ,olleH" （逆順）
```

### 1.2 文字列の結合と繰り返し

```python
# 文字列の結合
s1 = "Hello"
s2 = "World"
s1 + ", " + s2  # "Hello, World"

# join メソッドを使った結合
", ".join(["Hello", "World"])  # "Hello, World"

# 文字列の繰り返し
"Ha" * 3  # "HaHaHa"
```

### 1.3 文字列のメソッド

```python
# 大文字・小文字変換
"hello".upper()      # "HELLO"
"HELLO".lower()      # "hello"
"hello world".title()  # "Hello World"
"hello world".capitalize()  # "Hello world"

# 文字列の検索
"hello world".find("world")    # 6
"hello world".find("python")   # -1 （見つからない場合）
"hello world".index("world")   # 6
# "hello world".index("python") # ValueError （見つからない場合は例外）

# 文字列の置換
"hello world".replace("world", "python")  # "hello python"

# 文字列の分割
"hello,world,python".split(",")  # ["hello", "world", "python"]
"hello world python".split()     # ["hello", "world", "python"]

# 空白の削除
"  hello  ".strip()    # "hello"
"  hello  ".lstrip()   # "hello  "
"  hello  ".rstrip()   # "  hello"

# 文字列の判定
"123".isdigit()      # True
"abc".isalpha()      # True
"abc123".isalnum()   # True
"hello".startswith("he")  # True
"hello".endswith("lo")    # True
```

## 2. 文字列操作の応用

### 2.1 文字列の逆順と回文判定

```python
# 文字列の逆順
def reverse_string(s):
    return s[::-1]

# 回文判定
def is_palindrome(s):
    return s == s[::-1]

# 例
reverse_string("hello")      # "olleh"
is_palindrome("level")       # True
is_palindrome("hello")       # False
```

### 2.2 部分文字列の数え上げ

```python
# 部分文字列の出現回数
def count_substring(s, sub):
    count = 0
    pos = 0
    while True:
        pos = s.find(sub, pos)
        if pos == -1:
            break
        count += 1
        pos += 1  # 重複を許す場合
        # pos += len(sub)  # 重複を許さない場合
    return count

# 例
count_substring("banana", "ana")  # 1 (重複を許さない場合)
                                  # 2 (重複を許す場合)
```

### 2.3 文字の出現頻度

```python
from collections import Counter

# 文字の出現頻度をカウント
def char_frequency(s):
    return Counter(s)

# 例
char_frequency("banana")  # Counter({'a': 3, 'n': 2, 'b': 1})
```

### 2.4 アナグラム判定

```python
# 二つの文字列がアナグラムかどうか判定
def is_anagram(s1, s2):
    return sorted(s1) == sorted(s2)

# または
def is_anagram_counter(s1, s2):
    return Counter(s1) == Counter(s2)

# 例
is_anagram("listen", "silent")  # True
is_anagram("hello", "world")    # False
```

## 3. 文字列パターンマッチング

### 3.1 単純な文字列検索

```python
# 文字列内で部分文字列を検索
def string_search(text, pattern):
    n, m = len(text), len(pattern)
    for i in range(n - m + 1):
        if text[i:i+m] == pattern:
            return i  # 最初に見つかった位置
    return -1  # 見つからなかった場合

# 例
string_search("hello world", "world")  # 6
```

### 3.2 KMP法（Knuth-Morris-Pratt Algorithm）

効率的な文字列検索アルゴリズム。前回の比較情報を利用して不要な比較をスキップします。

```python
def kmp_search(text, pattern):
    n, m = len(text), len(pattern)
    if m == 0:
        return 0
    
    # 部分一致テーブルの構築
    lps = [0] * m  # longest proper prefix which is also suffix
    i = 1
    length = 0
    while i < m:
        if pattern[i] == pattern[length]:
            length += 1
            lps[i] = length
            i += 1
        else:
            if length != 0:
                length = lps[length - 1]
            else:
                lps[i] = 0
                i += 1
    
    # 文字列検索
    i = j = 0
    while i < n:
        if pattern[j] == text[i]:
            i += 1
            j += 1
        if j == m:
            return i - j  # パターンが見つかった位置
        elif i < n and pattern[j] != text[i]:
            if j != 0:
                j = lps[j - 1]
            else:
                i += 1
    return -1  # パターンが見つからなかった場合
```

### 3.3 Rabin-Karp法

ハッシュを用いた文字列検索アルゴリズム。高速に複数の部分文字列を探せます。

```python
def rabin_karp(text, pattern):
    n, m = len(text), len(pattern)
    if m == 0:
        return 0
    
    # 素数とベース（通常はアルファベットの数より大きい）
    p = 31
    mod = 10**9 + 9
    
    # パターンのハッシュ値を計算
    pattern_hash = 0
    for char in pattern:
        pattern_hash = (pattern_hash * p + ord(char)) % mod
    
    # テキストの各m文字のハッシュ値を計算
    text_hash = 0
    p_pow = [1]
    for i in range(m):
        text_hash = (text_hash * p + ord(text[i])) % mod
        if i < m - 1:
            p_pow.append((p_pow[-1] * p) % mod)
    
    # パターンとテキストのハッシュ値が一致する場所を探す
    if pattern_hash == text_hash and text[:m] == pattern:
        return 0
    
    for i in range(1, n - m + 1):
        # ローリングハッシュで次のハッシュ値を計算
        text_hash = (text_hash - ord(text[i-1]) * p_pow[m-1]) % mod
        text_hash = (text_hash * p + ord(text[i+m-1])) % mod
        if text_hash < 0:
            text_hash += mod
        
        # ハッシュ値が一致したら文字列も比較
        if pattern_hash == text_hash and text[i:i+m] == pattern:
            return i
    
    return -1
```

## 4. 動的計画法による文字列処理

### 4.1 最長共通部分列（LCS）

```python
def longest_common_subsequence(s, t):
    n, m = len(s), len(t)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if s[i-1] == t[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    
    return dp[n][m]

# 例
longest_common_subsequence("abcde", "ace")  # 3 ("ace")
```

### 4.2 編集距離（Levenshtein Distance）

```python
def edit_distance(s, t):
    n, m = len(s), len(t)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    
    # 初期化
    for i in range(n + 1):
        dp[i][0] = i
    for j in range(m + 1):
        dp[0][j] = j
    
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            cost = 0 if s[i-1] == t[j-1] else 1
            dp[i][j] = min(
                dp[i-1][j] + 1,      # 削除
                dp[i][j-1] + 1,      # 挿入
                dp[i-1][j-1] + cost  # 置換
            )
    
    return dp[n][m]

# 例
edit_distance("kitten", "sitting")  # 3
```

### 4.3 最長回文部分文字列

```python
def longest_palindromic_substring(s):
    n = len(s)
    if n == 0:
        return ""
    
    # dp[i][j] = s[i:j+1]が回文かどうか
    dp = [[False] * n for _ in range(n)]
    
    start = 0
    max_length = 1
    
    # 長さ1の回文
    for i in range(n):
        dp[i][i] = True
    
    # 長さ2の回文
    for i in range(n - 1):
        if s[i] == s[i+1]:
            dp[i][i+1] = True
            start = i
            max_length = 2
    
    # 長さ3以上の回文
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j] and dp[i+1][j-1]:
                dp[i][j] = True
                if length > max_length:
                    start = i
                    max_length = length
    
    return s[start:start + max_length]

# 例
longest_palindromic_substring("babad")  # "bab" または "aba"
longest_palindromic_substring("cbbd")   # "bb"
```

## 5. Trie（トライ木）

プレフィックス検索などに効率的なデータ構造です。

```python
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end_of_word = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end_of_word = True
    
    def search(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end_of_word
    
    def starts_with(self, prefix):
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True

# 使用例
trie = Trie()
words = ["apple", "banana", "application", "app"]
for word in words:
    trie.insert(word)

print(trie.search("apple"))       # True
print(trie.search("app"))         # True
print(trie.search("orange"))      # False
print(trie.starts_with("app"))    # True
print(trie.starts_with("ban"))    # True
print(trie.starts_with("cat"))    # False
```

## 6. まとめ

文字列処理のテクニックは競技プログラミングでよく出題される分野です。以下のポイントを抑えておくと良いでしょう：

1. 基本的な文字列操作（検索、置換、分割など）
2. 効率的なアルゴリズム（KMP法、Rabin-Karp法など）
3. 動的計画法を用いた文字列処理（LCS、編集距離など）
4. 特殊なデータ構造（Trie）の活用

これらのテクニックを組み合わせることで、様々な文字列問題に対応することができます。
